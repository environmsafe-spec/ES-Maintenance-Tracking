rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }
    function isAdmin() {
      return isSignedIn() &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Every collection follows the same pattern: any signed-in account can read (so a
    // "viewer" account can see everything), only an admin account can write. This
    // includes the `users` collection itself — email/role are not sensitive, and only
    // an admin can create, promote, or demote accounts. Passwords are never stored here;
    // Firebase Authentication holds those separately and this app never sees them.
    match /users/{userId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /generators/{doc} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /readings/{doc} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /maintenance/{doc} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /corrective/{doc} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /servicelog/{doc} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /settings/{doc} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /customfields/{doc} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
  }
}
