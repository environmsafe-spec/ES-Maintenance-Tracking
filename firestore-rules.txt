rules_version = '2';
// Only signed-in, NON-anonymous users (email/password accounts) may read or write.
// Anonymous tokens can be minted by anyone with the public API key, so they are rejected.
// Every collection the app uses must be listed here; anything else is denied by the
// catch-all at the bottom.
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null
        && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }
    match /generators/{doc} {
      allow read, write: if signedIn();
    }
    match /readings/{doc} {
      allow read, write: if signedIn();
    }
    match /maintenance/{doc} {
      allow read, write: if signedIn();
    }
    match /corrective/{doc} {
      allow read, write: if signedIn();
    }
    match /servicelog/{doc} {
      allow read, write: if signedIn();
    }
    match /settings/{doc} {
      allow read, write: if signedIn();
    }
    match /customfields/{doc} {
      allow read, write: if signedIn();
    }
    match /pricelist/{doc} {
      allow read, write: if signedIn();
    }
    match /invoices/{doc} {
      allow read, write: if signedIn();
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
